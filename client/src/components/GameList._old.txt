// baseball-ui/src/components/GameList.tsx
import { useEffect, useState, type ReactElement } from 'react';
import { fetchGamesByDate } from '../api';
import type { GameRow } from '../types';

export function GameList({
  date, selected, onSelect,
}: { date: string; selected?: string; onSelect: (gameId: string) => void }): ReactElement {
  const [games, setGames] = useState<GameRow[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    setLoading(true);
    fetchGamesByDate(date).then(setGames).finally(() => setLoading(false));
  }, [date]);

  if (loading) return <p>Loading games for {date}…</p>;
  if (!games.length) return <p>No games on {date}.</p>;

  return (
    <div style={{ display: 'grid', gap: 8 }}>
      {games.map((g): ReactElement => (
        <div key={g.providerGameId}
             style={{ border: '1px solid #ddd', padding: 12, borderRadius: 8,
                      background: selected === g.providerGameId ? '#f6faff' : 'white' }}>
          <div style={{ fontWeight: 600, color: '#333' }}>
            {g.awayAbbr} @ {g.homeAbbr} <small>({g.status})</small>
          </div>
          <button onClick={() => onSelect(g.providerGameId)} style={{ marginTop: 6 }}>
            {selected === g.providerGameId ? 'Listening…' : 'Join live'}
          </button>
        </div>
      ))}
    </div>
  );
}
